@using Service.Shared.Commons.Interfaces.Extentions
@using Service.Shared.Commons.Model.ServiceCustomHttpClient
@using Service.Shared.Contracts.Portal.DTOs
@using Service.Shared.Contracts.Portal.Forms
@using Service.UI.CMS.Blazor.Applications.Dtos
@using Microsoft.FluentUI.AspNetCore.Components
@using Microsoft.FluentUI.AspNetCore.Components.Icons.Filled

@inherits ComponentBase
@implements IDialogContentComponent<EditOrUpdateParametersDto>

<FluentDialogHeader ShowDismiss="true">
    <FluentStack VerticalAlignment="VerticalAlignment.Center">
        <FluentIcon Value="@(new Icons.Regular.Size24.Info())" />
        <FluentLabel Typo="Typography.PaneHeader">
            @Dialog.Instance.Parameters.Title
        </FluentLabel>
    </FluentStack>
</FluentDialogHeader>


@if (!IsLoading)
{


    <FluentDialogBody>
        <FluentGrid Class="s-form-view" Spacing="2" AdaptiveRendering="true" Justify="JustifyContent.FlexStart" Style="padding: 4px;">
            
            <FluentGridItem md="12">
                <div class="row-info">
                    <label>Ngày tạo:</label>
                    <div>@Item.Created.ToString("HH:mm:ss dd/MM/yyyy")</div>
                </div>
            </FluentGridItem>
            <FluentGridItem md="12">
                <div class="row-info">
                    <label>Ngày sửa cuối:</label>
                    <div>@Item.LastModified.ToString("HH:mm:ss dd/MM/yyyy ")</div>
                </div>
            </FluentGridItem>
            <FluentGridItem md="12">
                <div class="row-info">
                    <label>Trạng thái:</label>
                    <div>@(Item.ModerationStatus != 0 ? "Chưa duyệt" : "Đã duyệt")</div>
                </div>
            </FluentGridItem>
        </FluentGrid>
    </FluentDialogBody>
}
else
{
    <FluentDialogBody>
        <FluentProgressRing />
    </FluentDialogBody>
}


<FluentDialogFooter>
    <FluentButton Appearance="Appearance.Neutral" OnClick="CancelAsync">Đóng</FluentButton>
</FluentDialogFooter>




@code {
    [Inject]
    private IDialogService DialogService { get; set; } = default!;

    [Inject]
    private ICallServiceRegistry CallService { get; set; } = default!;

    [Inject]
    private IToastService ToastService { get; set; } = default!;


    [CascadingParameter]
    public FluentDialog Dialog { get; set; } = default!;

    [Parameter]
    public EditOrUpdateParametersDto Content { get; set; } = new EditOrUpdateParametersDto();

    private @Entity@Dto Item { get; set; } = new();
    private bool IsLoading { get; set; } = true;


    protected override async Task OnAfterRenderAsync(bool firtsLoad)
    {
        if (firtsLoad)
        {
            try
            {
                if (Content.Id != Guid.Empty)
                    Item = await FetchDataById(Content.Id);
                else
                    Item = new();

            }
            catch (Exception ex)
            {
                ToastService.ShowWarning($"Lỗi khi tải dữ liệu chi tiết: {ex.Message}");
            }
            finally
            {
                IsLoading = false;
            }
            StateHasChanged();
        }

    }

    private async Task<@Entity@Dto> FetchDataById(Guid Id)
    {
        ApiRequestModel apiRequest = new ApiRequestModel()
        {

            ApiService = ServicesRegistryEnum.ServicePortal,
            Endpoint = $"/@Entity@/{Id}"

        };

        ResultAPI<@Entity@Dto> result = await CallService.Get<@Entity@Dto>(apiRequest);
        if (result.Status != StatusCode.OK || result.Data is null)
        {
            ToastService.ShowWarning("Có lỗi khi gọi api lấy dữ liệu");
            return new();
        }
        return result.Data;

    }


    private async Task CancelAsync()
    {
        await Dialog.CancelAsync();
    }
}
