<script>
   
    function xem@Entity@(Id) {
        openModalWithApi("/@Entity@/View/" + Id, "lg");
    };
    function sua@Entity@(Id) {
        ChuyenHuong("external", "/@Entity@/Edit/" + Id);
    };

    function duyet@Entity@(Id) {
        toggleButtonContainerLoading(Id);
        $.ajax({
            url: '/@Entity@/Approve/' + Id,
            method: 'GET',
            dataType: 'json',
            success: function (data) {
                showNotification("success", "Duyệt thành công");
                toggleButtonContainerLoading(Id);
                $('#search-btn').click();
            },
            error: function (xhr, status, error) {
                HandleErrorRespone(xhr);
                toggleButtonContainerLoading(Id);
            }
        });
    };
    function huyDuyet@Entity@(Id) {
        toggleButtonContainerLoading(Id);
        $.ajax({
            url: '/@Entity@/Reject/' + Id,
            method: 'GET',
            dataType: 'json',
            success: function (data) {
                toggleButtonContainerLoading(Id);
                showNotification("success", "Hủy duyệt thành công")

                $('#search-btn').click();
            },
            error: function (xhr, status, error) {
                toggleButtonContainerLoading(Id);
                HandleErrorRespone(xhr);
            }
        });
    };
    function xoa@Entity@(Id) {

        toggleButtonContainerLoading(Id);
        showNotificationWithConfirm(
            "",
            function () {
                $.ajax({
                    url: '/@Entity@/Delete/' + Id,
                    method: 'GET',
                    dataType: 'json',
                    success: function (data) {
                        toggleButtonContainerLoading(Id);
                        showNotification("warrning", "Xóa thành công")
                        $('#search-btn').click();
                    },
                    error: function (xhr, status, error) {
                        console.log(error)
                        HandleErrorRespone(xhr);
                        toggleButtonContainerLoading(Id);
                    }
                });
            },
            function () {
                toggleButtonContainerLoading(Id);
            });

    };
    $(document).ready(function () {


        var dataGridInstance = $('#data-source-4').registerGrid({
            "columns": [
                {
                    "data": 'STT',
                    "title": "STT",
                    "render": function (data, type, row, meta) {
                        return Index = meta.row + meta.settings._iDisplayStart + 1;
                    },
                    "orderable": false,
                    "visible": true,
                    "width": "50px"
                },
                @columnsTemplate@
                

            ],
            length: 20,
            endpoint: "/@Entity@/GetPaged",
            prepareRequest: function (d) {
                d.SearchIn = ["TenChuDe"];
                d.Keyword = $('#search-input').val();
                var t_gridRequest = {
                    "skip": d.start,
                    "page": (d.start / d.length) + 1,
                    "pageSize": d.length
                };
                d.t_gridRequest = JSON.stringify(t_gridRequest);
                return d;
            },

            // dataBound: function () {
            //     // $('.dataTables_processing').hide();
            //     // console.log("load xong dữ liệu")
            // },

        });
        $('#search-btn').registerButton({
            type: 'primary',
            icon: "fa fa-search",
            clickHandler: function () {

                return new Promise(function (resolve) {
                    var grid = dataGridInstance;
                    grid.page(0);//chuyển về trang đầu
                    grid.ajax.reload(null, false);
                    resolve();//cờ xác định kết thúc
                });
            }
        });
        $('#exportExcel').registerButton({
            type: 'info',
            icon: "fa fa-file-excel-o",
            clickHandler: function () {
                var grid = dataGridInstance;
                return new Promise(function (resolve, reject) {
                    $.post("/@Entity@/ExportDataToExcel", {
                        "SearchIn": ["Ten@Entity@", "ThongTin"],
                        "Keyword": $('#search-input').val()
                    })
                        .done(function (data) {
                            console.log(data)
                            ChuyenHuong("external", `/Files/ExportExcel?file=${data}`, "new_tab")
                            resolve(); // Kết thúc promise
                        })
                        .fail(function (xhr, status, error) {
                            console.log(xhr);
                            HandleErrorRespone(xhr);
                            reject(error); // Kết thúc promise với lỗi
                        });
                });

            }
        });

        $("#search-input").enterKey(function () {
            var button = $('#search-btn');
            if (!button.hasClass("loading"))
                $('#search-btn').click();
        });

        $('#datetime-local').datepicker({
            language: 'vi',

        })
        $('#Them@Entity@').registerButton({
            type: 'primary',
            icon: "fa fa-plus",
            clickHandler: function () {
                return new Promise(function (resolve) {
                    ChuyenHuong("external", "/@Entity@/Edit/" + 0)
                    resolve();//cờ xác định kết thúc
                });
            }
        });


    });
</script>
